// @vitest-environment jsdom
import { act, cleanup, fireEvent, render, screen, waitFor, within } from '@testing-library/react'
import { afterEach, expect, it, vi } from 'vitest'
import { AssistantPanel } from '../../src/renderer/src/features/assistants/AssistantPanel'
import type { AssistantApi, AssistantSnapshot, AssistantResult } from '../../src/shared/assistant-contract'

afterEach(cleanup)
const id = '11111111-1111-4111-8111-111111111111'
function snapshot(revision: number, persona = 'old', deleted = false): AssistantSnapshot {
  return {
    stateRevision: revision,
    currentAssistantId: deleted ? null : id,
    primaryAssistantId: deleted ? null : id,
    assistants: deleted ? [] : [{
      id, displayName: 'Reviewer', persona, avatarKey: 'leaf', isArchived: false,
      createdAt: '2026-09-07T00:00:00.000Z', updatedAt: '2026-09-07T00:00:00.000Z',
      archivedAt: null, version: revision
    }]
  }
}

it.each(['equal-clone', 'newer-delete'] as const)('independent: %s snapshot during save preserves only the valid completion', async kind => {
  const initial = snapshot(1)
  let resolveSave!: (result: AssistantResult) => void
  const api: AssistantApi = {
    list: vi.fn(async () => ({ ok: true, data: initial })),
    rename: vi.fn(() => new Promise(resolve => { resolveSave = resolve })),
    create: vi.fn(), switch: vi.fn(), archive: vi.fn(), setPrimary: vi.fn()
  }
  const notified = vi.fn()
  const view = render(<AssistantPanel api={api} externalSnapshot={initial} onSnapshot={notified} />)
  const card = await screen.findByRole('listitem', { name: '助手配置：Reviewer' })
  await waitFor(() => expect(api.list).toHaveBeenCalledTimes(1))
  fireEvent.change(within(card).getByRole('textbox', { name: '人设 Reviewer' }), { target: { value: 'new-reviewed-profile' } })
  fireEvent.click(within(card).getByRole('button', { name: '保存基础配置' }))
  await waitFor(() => expect(api.rename).toHaveBeenCalledTimes(1))
  const external = kind === 'equal-clone' ? structuredClone(initial) : snapshot(3, '', true)
  view.rerender(<AssistantPanel api={api} externalSnapshot={external} onSnapshot={notified} />)
  await act(async () => { await Promise.resolve() })
  if (kind === 'newer-delete') expect(screen.queryByRole('textbox', { name: '人设 Reviewer' })).not.toBeInTheDocument()
  await act(async () => { resolveSave({ ok: true, data: snapshot(2, 'new-reviewed-profile') }); await Promise.resolve() })
  if (kind === 'equal-clone') {
    expect(await screen.findByText('已保存 Reviewer 的基础配置（版本 2）')).toBeInTheDocument()
    expect(screen.getByRole('textbox', { name: '人设 Reviewer' })).toHaveValue('new-reviewed-profile')
    expect(notified).toHaveBeenCalledWith(snapshot(2, 'new-reviewed-profile'))
  } else {
    expect(screen.queryByRole('listitem', { name: '助手配置：Reviewer' })).not.toBeInTheDocument()
    expect(screen.queryByText('已保存 Reviewer 的基础配置（版本 2）')).not.toBeInTheDocument()
    expect(notified).not.toHaveBeenCalledWith(snapshot(2, 'new-reviewed-profile'))
  }
})
